$ cat /tmp/plugin-fingerprint.py
from pathlib import Path
import hashlib
import sys
reference, installed = Path(sys.argv[1]), Path(sys.argv[2])
def files(root):
    return sorted(str(p.relative_to(root)) for p in root.rglob("*")
                  if p.is_file() and ".git" not in p.relative_to(root).parts)
def fingerprint(root, names):
    h = hashlib.sha256()
    for name in names:
        h.update(name.encode() + b"\0" + hashlib.sha256((root / name).read_bytes()).hexdigest().encode() + b"\n")
    return h.hexdigest()
ref_files, inst_files = files(reference), files(installed)
print("reference:", reference)
print("installed:", installed)
print("paths only in reference tree:", sorted(set(ref_files) - set(inst_files)))
print("paths only in installed copy:", sorted(set(inst_files) - set(ref_files)))
a, b = fingerprint(installed, ref_files), fingerprint(reference, ref_files)
print("installed tree SHA256:", a)
print("reference tree SHA256:", b)
assert set(ref_files) == set(inst_files) and a == b
print("all", len(ref_files), "reference paths match the installed copy")
