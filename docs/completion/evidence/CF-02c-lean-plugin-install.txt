from pathlib import Path
import hashlib
cache = Path("/home/witness/.claude/plugins/cache/orca-fleet/orca-fleet/0.7.1")
dist = Path("/home/witness/dist-tree")
def files(root):
    return sorted(str(p.relative_to(root)) for p in root.rglob("*")
                  if p.is_file() and ".git" not in p.relative_to(root).parts)
def fingerprint(root, names):
    h = hashlib.sha256()
    for name in names:
        h.update(name.encode() + b"\0" + hashlib.sha256((root / name).read_bytes()).hexdigest().encode() + b"\n")
    return h.hexdigest()
dist_files, cache_files = files(dist), files(cache)
print("paths only in dist tree:", sorted(set(dist_files) - set(cache_files)))
print("paths only in installed copy (bookkeeping/generated):", sorted(set(cache_files) - set(dist_files)))
a, b = fingerprint(cache, dist_files), fingerprint(dist, dist_files)
print("installed tree SHA256:", a)
print("published dist tree SHA256:", b)
assert a == b
print("all", len(dist_files), "dist paths match the installed copy")
